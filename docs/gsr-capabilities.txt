### comando: flatpak run --command=gsr-cli com.dec05eba.gpu_screen_recorder --help
### codigo: 0
usage: gsr-cli -ipc <socket_path> <command> [command_arguments...]

Sends a command to a GPU Screen Recorder instance that was started with the -ipc option.

OPTIONS:
  -ipc <socket_path>
    The unix domain socket that GPU Screen Recorder was started with. Required.

COMMANDS:
  status
    Check if a GPU Screen Recorder instance is listening on the socket. Prints "running" or
    "not running" and exits with 0 when it's running.
  stop
    Stop and save the recording (stop without save in replay mode). Waits until the recording
    has been saved and prints the path of the saved file.
  toggle-pause
    Pause/unpause the recording (not for streaming/replay).
  set-paused true|false
    Pause/unpause the recording (not for streaming/replay). Unlike toggle-pause this doesn't
    fail when the recording is already paused/unpaused.
  toggle-replay-recording
    Start/stop a regular recording during replay/streaming.
  start-replay-recording
    Start a regular recording during replay/streaming. Does nothing when a recording is already running.
  stop-replay-recording
    Stop the regular recording that runs during replay/streaming. Waits until the recording
    has been saved and prints the path of the saved file.
  save-replay [seconds] [restart-replay=true|false]
    Save the replay. The number of seconds has to be larger than 0. The whole replay buffer is
    saved when no number of seconds is given. Waits until the replay has been saved and prints
    the path of the saved file. restart-replay overrides the -restart-replay-on-save option
    of GPU Screen Recorder for this save.

EXAMPLES:
  gsr-cli -ipc "$XDG_RUNTIME_DIR/gsr.sock" status
  gsr-cli -ipc "$XDG_RUNTIME_DIR/gsr.sock" save-replay 30
  gsr-cli -ipc "$XDG_RUNTIME_DIR/gsr.sock" save-replay restart-replay=true
